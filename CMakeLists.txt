function(chi_strict target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
    # Sol: MSVC14.51 incremental linker crashes after native data-layout edits
    # (LNK1000 IncrBuildImage). Full executable links retain Debug information.
    get_target_property(chi_target_kind ${target} TYPE)
    if(chi_target_kind STREQUAL "EXECUTABLE")
      target_link_options(${target} PRIVATE /INCREMENTAL:NO)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    if(CHI_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
  endif()
  # Test assertions are acceptance checks, not optional Debug diagnostics.
  # Keep production Release flags intact while testing optimized game methods.
  if(target MATCHES "^test_")
    if(MSVC)
      target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
  endif()
endfunction()
